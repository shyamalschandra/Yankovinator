\begin{frame}[fragile]
\frametitle{How It Works}
\begin{enumerate}
    \item Analyze original song's syllable structure
    \item Detect rhyming scheme automatically
    \item Extract theme keywords and definitions
    \item Generate each line with:
    \begin{itemize}
        \item Exact syllable matching (word-by-word)
        \item Rhyming constraints
        \item Semantic coherence with previous lines
        \item Theme advancement
    \end{itemize}
    \item Refine for semantic coherence
    \item Apply capitalization and punctuation matching
    \item Maintain rhythm, style, and meaning
\end{enumerate}

\begin{lstlisting}
let parody = try await Yankovinator.generateParody(
    originalLyrics: lyrics,
    keywords: keywords
)
\end{lstlisting}
\end{frame}

\begin{frame}
\frametitle{Algorithm Flow}
\begin{enumerate}
    \item Parse input lyrics
    \item Analyze syllable structure (total and word-by-word)
    \item Detect rhyming scheme
    \item Load theme keywords
    \item For each line:
    \begin{itemize}
        \item Build prompt with constraints and context
        \item Request from Ollama with previous lines
        \item Validate syllable count
        \item Refine word-by-word matching
        \item Refine semantic coherence
        \item Apply capitalization and punctuation
        \item Clean and format
    \end{itemize}
    \item Return complete parody
\end{enumerate}
\end{frame}
